cmake_minimum_required(VERSION 3.24)
project(signalisationfrancaiserealiste VERSION 0.1.0 LANGUAGES CXX)
find_package(NimbyRailsFranceSDK 0.7.1 CONFIG REQUIRED
  HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/install/development")
add_custom_target(sfr_test_textures ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/imgs" "$<TARGET_FILE_DIR:sfr_mod>/imgs")
add_library(sfr_mod SHARED src/mod.cpp)
target_compile_features(sfr_mod PRIVATE cxx_std_20)
target_compile_definitions(sfr_mod PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(sfr_mod PROPERTIES PREFIX "" OUTPUT_NAME "SignalisationFrancaiseRealisteMod")
target_link_libraries(sfr_mod PRIVATE NimbyRailsFranceSDK::Mod)
if(MINGW)
  target_link_options(sfr_mod PRIVATE -static-libgcc -static-libstdc++)
endif()
install(TARGETS sfr_mod RUNTIME DESTINATION .)
install(FILES assets/mod.txt assets/nrf-mod.ini DESTINATION .)
install(DIRECTORY imgs DESTINATION .)
if(MSVC)
  target_compile_options(sfr_mod PRIVATE /utf-8)
endif()
add_custom_command(TARGET sfr_mod POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    ${NimbyRailsFranceSDK_RUNTIME_FILES} "$<TARGET_FILE_DIR:sfr_mod>"
  COMMAND_EXPAND_LISTS)
enable_testing()
add_executable(sfr_texture_assets_test tests/textures.cpp)
target_compile_features(sfr_texture_assets_test PRIVATE cxx_std_20)
add_dependencies(sfr_texture_assets_test sfr_test_textures)
if(MINGW)
  target_link_options(sfr_texture_assets_test PRIVATE -static-libgcc -static-libstdc++)
endif()
add_test(NAME bal_a_textures COMMAND sfr_texture_assets_test)
set(SFR_LOADER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../sdk" CACHE PATH "SDK sources for NRF Loader integration test")
if(EXISTS "${SFR_LOADER_SOURCE_DIR}/src/loader/mods.cpp")
  add_executable(sfr_mod_loader_test tests/mod_loader.cpp "${SFR_LOADER_SOURCE_DIR}/src/loader/mods.cpp")
  target_compile_features(sfr_mod_loader_test PRIVATE cxx_std_20)
  target_include_directories(sfr_mod_loader_test PRIVATE "${SFR_LOADER_SOURCE_DIR}/include")
  target_compile_definitions(sfr_mod_loader_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MINGW)
    target_link_options(sfr_mod_loader_test PRIVATE -static-libgcc -static-libstdc++)
  endif()
  add_dependencies(sfr_mod_loader_test sfr_mod)
  add_test(NAME nrf_mod_autoload COMMAND sfr_mod_loader_test "$<TARGET_FILE:sfr_mod>")
endif()
